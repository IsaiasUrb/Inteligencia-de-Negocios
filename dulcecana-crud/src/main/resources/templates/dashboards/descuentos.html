Ver consultas SQL usadas 
SELECT count(*) FILTER (WHERE descuento_pct > 0) AS clientes_con_descuento, count(*) AS total_clientes
FROM public.cliente;

SELECT round(avg(descuento_pct), 2) AS descuento_promedio
FROM public.cliente WHERE descuento_pct > 0;

SELECT nombre, descuento_pct FROM public.cliente ORDER BY descuento_pct DESC LIMIT 1;

SELECT COALESCE(SUM(fv.subtotal), 0) AS ingreso_clientes_descuento
FROM dw_dulce_cana.fact_ventas_dc fv
JOIN dw_dulce_cana.dim_cliente_dc dc ON dc.id_cliente_dw = fv.id_cliente_dw
WHERE dc.descuento_pct > 0;

SELECT id_cliente, nombre, tipo_cliente, descuento_pct
FROM public.vw_descuentos_cliente;
