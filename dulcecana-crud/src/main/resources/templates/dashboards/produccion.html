SELECT anio, mes, nombre_mes, costo_materia_prima, costo_mano_obra, costo_indirecto, costo_total
FROM dw_dulce_cana.vw_costos_mensuales ORDER BY anio, mes;

SELECT id_lote_origen, fecha_produccion, proveedor, cantidad_kg_producida, costo_total,
       ingreso_atribuido, utilidad_bruta, margen_utilidad_pct, rendimiento_pct
FROM dw_dulce_cana.vw_rentabilidad_lote ORDER BY fecha_produccion DESC;
